# -----------------------------
# Function to calculate metrics
# -----------------------------
get_metrics <- function(actual, predicted) {
  # Keep only complete cases
  good_rows <- complete.cases(actual, predicted)
  actual <- actual[good_rows]
  predicted <- predicted[good_rows]

  error <- actual - predicted

  # Avoid division by zero for percentage errors
  nonzero <- actual != 0

  R2 <- cor(actual, predicted)^2

  MAE <- mean(abs(error))

  MAPE <- mean(abs(error[nonzero] / actual[nonzero])) * 100

  RAE <- sum(abs(error)) / sum(abs(actual - mean(actual))) * 100

  RMSE <- sqrt(mean(error^2))

  RMSPE <- sqrt(mean((error[nonzero] / actual[nonzero])^2)) * 100

  RRSE <- sqrt(sum(error^2) / sum((actual - mean(actual))^2)) * 100

  data.frame(
    R2 = R2,
    MAE = MAE,
    MAPE = MAPE,
    RAE = RAE,
    RMSE = RMSE,
    RMSPE = RMSPE,
    RRSE = RRSE
  )
}

1.3 Question 3: Cross-validation of lm, rpart, and M5P NA_Sales prediction models

1.3.1 Part A - Cross-validation function

# Named function for cross-validation of numeric prediction models
cv_numeric_model <- function(data, target, model_method, k = 10, seed = 123) {
  
  set.seed(seed)
  
  # Create k folds based on the target variable
  folds <- createFolds(data[[target]], k = k, list = TRUE, returnTrain = FALSE)
  
  # Function to calculate model-fit and error metrics
  get_metrics <- function(actual, predicted) {
    
    good_rows <- complete.cases(actual, predicted)
    actual <- actual[good_rows]
    predicted <- predicted[good_rows]
    
    error <- actual - predicted
    
    nonzero <- actual != 0
    
    R2 <- cor(actual, predicted)^2
    
    MAE <- mean(abs(error))
    
    MAPE <- mean(abs(error[nonzero] / actual[nonzero])) * 100
    
    RAE <- sum(abs(error)) / sum(abs(actual - mean(actual))) * 100
    
    RMSE <- sqrt(mean(error^2))
    
    RMSPE <- sqrt(mean((error[nonzero] / actual[nonzero])^2)) * 100
    
    RRSE <- sqrt(sum(error^2) / sum((actual - mean(actual))^2)) * 100
    
    data.frame(
      R2 = R2,
      MAE = MAE,
      MAPE = MAPE,
      RAE = RAE,
      RMSE = RMSE,
      RMSPE = RMSPE,
      RRSE = RRSE
    )
  }
  
  # Store results for each fold
  fold_results <- data.frame()
  
  for (i in seq_along(folds)) {
    
    test_index <- folds[[i]]
    
    cv_train <- data[-test_index, ]
    cv_test <- data[test_index, ]
    
    formula_text <- paste(target, "~ .")
    model_formula <- as.formula(formula_text)
    
    # Train the selected model using default settings
    if (model_method == "lm") {
      model <- lm(model_formula, data = cv_train)
    } else if (model_method == "rpart") {
      model <- rpart(model_formula, data = cv_train)
    } else if (model_method == "M5P") {
      model <- M5P(model_formula, data = cv_train)
    } else {
      stop("model_method must be 'lm', 'rpart', or 'M5P'")
    }
    
    # Predict on the validation fold
    predicted <- predict(model, newdata = cv_test)
    
    # Calculate fold metrics
    fold_metrics <- get_metrics(cv_test[[target]], predicted)
    
    fold_results <- bind_rows(
      fold_results,
      cbind(
        Model = model_method,
        Fold = i,
        fold_metrics
      )
    )
  }
  
# Calculate means across folds
metric_means <- fold_results %>%
  summarise(
    across(
      c(R2, MAE, MAPE, RAE, RMSE, RMSPE, RRSE),
      \(x) mean(x, na.rm = TRUE)
    )
  ) %>%
  mutate(Model = model_method, Fold = "Mean") %>%
  select(Model, Fold, everything())

# Calculate standard deviations across folds
metric_sds <- fold_results %>%
  summarise(
    across(
      c(R2, MAE, MAPE, RAE, RMSE, RMSPE, RRSE),
      \(x) sd(x, na.rm = TRUE)
    )
  ) %>%
  mutate(Model = model_method, Fold = "SD") %>%
  select(Model, Fold, everything())

# Convert Fold to character before binding with Mean and SD rows
fold_results <- fold_results %>%
  mutate(Fold = as.character(Fold))

# Combine fold results, means, and standard deviations
cv_results <- bind_rows(
  fold_results,
  metric_means,
  metric_sds
)
  
  return(cv_results)
}

2 Task 2: Reflections

  1. Predictor recommended for removal for parsimony

To keep the models more parsimonious, I would recommend removing User_Score first. It has the weakest correlation with NA_Sales among the numeric predictors, with a correlation of about 0.145, while Critic_Score, Critic_Count, and User_Count all have stronger relationships with sales. In the linear regression model, User_Score was statistically significant, but its coefficient was small and negative, about -0.0028. This is harder to interpret because higher user scores would normally be expected to relate to higher sales. Since User_Score also overlaps conceptually with Critic_Score, it may not add much unique information once the other predictors are already included. For that reason, I would remove User_Score before removing stronger and more interpretable predictors.

  1. Why log_User_Count could be more effective than User_Count

The log of User_Count could be more effective because the original User_Count variable is highly right-skewed. In this data set, the median User_Count is 24, the mean is about 152.7, and the maximum is 10,665. This shows that a small number of games have extremely high user counts. These large values can have too much influence on a linear model.

Using log_User_Count compresses the large values and better represents diminishing returns. For example, the difference between 10 and 100 user ratings probably matters more than the difference between 5,010 and 5,100 ratings. The model results also support this transformation. The original linear regression model had an R-squared of about 0.279, while the model using log_User_Count had an R-squared of about 0.363. This suggests that the log transformation helped explain more variation in NA_Sales. If the cross-validation table also shows lower MAE, RMSE, RAE, or RRSE for the log model, then that would further support using log_User_Count instead of the original User_Count.

  1. Whether to add a quadratic term of User_Count

I would not recommend adding a quadratic term of User_Count as my first choice. Even though a squared term could capture a curved relationship, User_Count is already extremely skewed. Since the maximum value is 10,665 and the median is only 24, squaring this variable would make the largest observations even more extreme. This could make the model more sensitive to a small number of very popular games.

A log transformation is more appropriate because it reduces skewness instead of making it worse. The log model also improved the linear model R-squared more than the original model. A quadratic User_Count model may improve fit, but it is less interpretable and less parsimonious. I would only use User_Count_Squared if cross-validation clearly showed better prediction errors than the log model. Based on the data exploration, I would recommend log_User_Count over User_Count_Squared.

  1. What I learned from the models and adjustments

Building these models showed me that each method has different strengths. The linear regression model was the easiest to interpret because it showed the direction, size, and significance of each predictor. It showed that variables like Critic_Score, Critic_Count, and User_Count were important, but the original model only explained about 28% of the variation in North American sales. This means the model is useful, but many sales factors are still missing, such as marketing, release timing, brand popularity, and whether a game belongs to a major franchise.

The rpart model was useful because it could capture non-linear patterns and interactions without requiring me to manually create them. However, a default decision tree can sometimes be too simple, so it may not always predict as well as other models. The M5P model was helpful because it combines decision-tree logic with linear regression models, so it can handle different groups of games in a more flexible way.

The quadratic term for Critic_Score improved the linear model fit, with R-squared increasing from about 0.279 to about 0.305. This suggests that the relationship between critic score and sales is not completely linear. The log transformation of User_Count had an even larger effect, increasing R-squared to about 0.363, which shows that transforming skewed variables can improve a model. If I were explaining this to a supervisor, I would say that the best model should not be chosen only by training fit. I would compare cross-validation RMSE and MAE and choose the model that predicts well while still being understandable. Overall, the log transformation of User_Count seems like the most useful adjustment because it improves the model and makes sense based on the distribution of the data.
